\gretlfpycmd{\_\_copy\_\_()}
\estdesc{Creates a deep copy of the \classpydataset object.}
\begin{lstlisting}[style=pythonstyle]
import copy
dataset_copy = copy.copy(gretl._gretl.Dataset)
\end{lstlisting}
\generaldesc{This method returns a new, fully independent
	\pythcode{gretl.\_gretl.Dataset} object that duplicates
	all elements of the original dataset. Changes made to
	the copy do not affect the original object. Although
	it is invoked by the standard \pythcode{copy.copy()}
	function, it performs a deep copy of all dataset
	contents.}

\gretlfpycmd{bkfilt()}
\estdesc{Returns the result from application of the Baxter--King
	bandpass filter to the series \textit{y}.}
\begin{lstlisting}[style=pythonstylefoot]
gretl._gretl.Dataset.bkfilt(series=str, f1=int, f2=int, k=int) -> numpy.ndarray
\end{lstlisting}
\maptof{bkfilt(y, f1, f2, k)}{series}
\generaldesc{The optional parameters \textit{f1} and
	\textit{f2} represent, respectively, the lower and
	upper bounds of the range of frequencies to extract,
	while \textit{k} is the approximation order to be used.}

\gretlfpycmd{bwfilt()}
\estdesc{Returns the result from application of a low-pass
	Butterworth filter with order \textit{n} and frequency
	cutoff \textit{omega} to the series \textit{y}.}
\begin{lstlisting}[style=pythonstylefoot]
gretl._gretl.Dataset.bwfilt(series=str, n=int, omega=float) -> numpy.ndarray
\end{lstlisting}
\maptof{bwfilt(y, n, omega)}{series}
\generaldesc{The cutoff is expressed in degrees and must be
	greater than 0 and less than 180. Smaller cutoff values
	restrict the pass-band to lower frequencies and hence
	produce a smoother trend. Higher values of \textit{n}
	produce a sharper cutoff, at the cost of possible
	numerical instability.}

Examples:\newline
\pythcode{.corr('y x z t')}\\
\pythcode{.corr('ylist', uniform=True)}\\
\pythcode{.corr('x y', spearman=True)}

\generaldesc{By default, prints the pairwise correlation
	coefficients (Pearson's product-moment correlation) for
	the variables in \textit{varlist}, or for all variables
	in the data set if \textit{varlist} is not given. The
	standard behavior is to use all available observations
	for computing each pairwise coefficient, but if the
	\gretlcmd{--uniform} option is given the sample is limited
	(if necessary) so that the same set of observations is
	used for all the coefficients. This option has an effect
	only if there are differing numbers of missing values
	for the variables used.}

\gretlfpycmd{hpfilt()}
\estdesc{Returns the cycle component from application of
	the Hodrick--Prescott filter to series \textit{y}.}
\begin{lstlisting}[style=pythonstylefoot]
gretl._gretl.Dataset.hpfilt(series=str, hp_lambda=float, one_sided=bool) -> numpy.ndarray
gretl._gretl.Dataset.hpfilt(series=str, hp_lambda=float, trend=bool, one_sided=bool) -> numpy.ndarray
\end{lstlisting}
\maptof{hpfilt(y, lambda, one-sided)}{series}
\generaldesc{If the smoothing parameter, \textit{lambda}%
	\footnote{Note the change of the \textit{lambda} parameter
		to \textit{hp\_lambda} in \gretlpy. This is because
		\pythcode{lambda} is a reserved keyword in \python.},
	is not supplied then a data-based default is used, namely
	100 times the square of the periodicity (100 for annual
	data, 1600 for quarterly data, and so on).\\
	By default the filter is the usual two-sided version,
	but if the optional third argument is given with a non-zero
	value a one-sided variant (with no look-ahead).\\
	The most common use of the HP filter is detrending, but if
	it's the trend you are interested in that is easily
	obtained by subtraction, as in:\newline
	\pythcode{hptrend = .hpfilt(y, trend=True)}.}

\gretlfpycmd{hurst()}
\estdesc{Calculates the Hurst exponent (a measure of persistence
	or long memory) for a time-series variable having at least 128 observations.}
\begin{lstlisting}[style=pythonstyle]
gretl._gretl.Dataset.hurst(series=str) -> numpy.ndarray
\end{lstlisting}
\mapto[clean]{hurst series}
\generaldesc{The Hurst exponent is discussed by Mandelbrot (1983).
	In theoretical terms it is the exponent, \textit{H}, in the relationship
	\[
		RS(x) = an^H,
	\]
	where \textit{RS} is the ``rescaled range'' of the variable
	\textit{x} in samples of size \textit{n} and \textit{a} is
	a constant. In gretl, the exponent is estimated using binary
	sub-sampling: we start with the entire data range, then the
	two halves of the range, then the four quarters, and so on.
	For sample sizes smaller than the data range, the \textit{RS}
	value is the mean across the available samples. The exponent
	is then estimated as the slope coefficient in a regression of
	the log of \textit{RS} on the log of sample size.}

\gretlfpycmd{sample()}
\estdesc{This command can be used only when a dataset is in place.}
\begin{lstlisting}[style=pythonstylefoot]
gretl._gretl.Dataset.sample(startobs=str, endobs=str, [OPTS])
gretl._gretl.Dataset.sample(startobs=str, [OPTS])
gretl._gretl.Dataset.sample(startobs=int, endobs=int, [OPTS])
gretl._gretl.Dataset.sample(startobs=int, [OPTS])
\end{lstlisting}
\mapto[clean]{smpl startobs endobss}
\mapto[clean]{smpl +i -j}
\mapto[clean]{smpl dumvar --dummy}
\mapto[clean]{smpl condition --restrict}
\mapto[clean]{smpl --no-missing [ varlist ]}
\mapto[clean]{smpl --no-all-missing [ varlist ]}
\mapto[clean]{smpl --contiguous [ varlist ]}
\mapto[clean]{smpl n --random}
\mapto[clean]{smpl full}
\mapto{smpl}
\begin{itemize}
	\item[]\maptoopt{dummy=True}{--dummy}{argument is a dummy variable}
	\item[]\maptoopt{restrict=True}{--restrict}{apply boolean restriction}
	\item[]\maptoopt{replace=True}{--replace}{replace any existing boolean restriction}
	\item[]\maptoopt{no\_missing=True}{--no-missing}{restrict to valid observations}
	\item[]\maptoopt{no\_all\_missing=True}{--no-all-missing}{omit empty observations (\seegretlmanual)}
	\item[]\maptoopt{contiguous=True}{--contiguous}{\seegretlmanual}
	\item[]\maptoopt{random=True}{--random}{form random sub-sample}
	\item[]\maptoopt{permanent=True}{--permanent}{\seegretlmanual}
	\item[]\maptoopt{preserve\_panel=True}{--preserve-panel}{\seegretlmanual}
	\item[]\maptoopt{unit=True}{--unit}{panel data: sample in cross-sectional dimension}
	\item[]\maptoopt{time=True}{--time}{panel data: sample in time-series dimension}
	\item[]\maptoopt{dates=True}{--dates}{interpret observation numbers as dates}
	\item[]\maptoopt{quiet=True}{--quiet}{don't report sample range}
\end{itemize}
Examples:\newline
\pythcode{.sample(3, 10)}\\
\pythcode{.sample('1960:2', '1982:4')}\\
\pythcode{.sample(1, -1)}\\
\pythcode{.sample('x > 3000', restrict=True)}\\
\pythcode{.sample('y > 3000', restrict=True, replace=True))}\\
\pythcode{.sample(100, random=True)}

\generaldesc{When no arguments are given, it displays the
	current sample range, otherwise it sets the sample range.
	The range can be defined in several ways. In the first
	alternate form (and the first two examples) above,
	\textit{startobs} and \textit{endobs} must be consistent
	with the periodicity of the data. Either one may be replaced
	by a semicolon to leave the value unchanged. (For more on
	\textit{startobs} and \textit{endobs} see the section titled
	``Dates versus sequential indices'' below.) In the second form,
	the integers \textit{i} and \textit{j} (which may be positive
	or negative, and must be signed) are taken as offsets relative
	to the existing sample range. In the third form \textit{dummyvar}
	must be an indicator variable with values 0 or 1 at each
	observation; the sample will be restricted to observations
	where the value is 1. The fourth form, using \gretlcmd{--restrict}
	restricts the sample to observations that satisfy the given Boolean condition.}

\gretlfpycmd{setobs()}
\estdesc{This command forces the program to interpret
	the current data set as having a specified structure.}
\begin{lstlisting}[style=pythonstylefoot]
gretl._gretl.Dataset.setobs(periodicity=int, startobs=str, [OPTS])
gretl._gretl.Dataset.setobs(periodicity=int, startobs=int, [OPTS])
gretl._gretl.Dataset.setobs(unitvar=str, timevar=str, [OPTS])
\end{lstlisting}
\mapto[clean]{setobs periodicity startobs}
\mapto{setobs unitvar timevar --panel-vars}
\begin{itemize}
	\item[]\maptoopt{cross\_section=True}{--cross-section}{interpret as cross section}
	\item[]\maptoopt{time\_series=True}{--time-series}{interpret as time series}
	\item[]\maptoopt{special\_time\_series=True}{--special-time-series}{\seegretlmanual}
	\item[]\maptoopt{stacked\_cross\_section=True}{--stacked-cross-section}{interpret as panel data}
	\item[]\maptoopt{stacked\_time\_series=True}{--stacked-time-series}{interpret as panel data}
	\item[]\maptoopt{panel\_vars=True}{--panel-vars}{\seegretlmanual}
	\item[]\maptoopt{panel\_time=True}{--panel-time}{\seegretlmanual}
	\item[]\maptoopt{panel\_groups=True}{--panel-groups}{\seegretlmanual}
\end{itemize}
Examples:\newline
\pythcode{.setobs(4, '1990:1', time\_series=True}\\
\pythcode{.setobs(12, '1978:03')}\\
\pythcode{.setobs(1, 1, cross\_section=True)}\\
\pythcode{.setobs(20, '1:1', stacked\_time\_series=True)}\\
\pythcode{.setobs(unit, year, panel\_vars=True)}

\generaldesc{In the first and in the second form of the method
	the periodicity, which must be an integer, represents frequency
	in the case of time-series data (1 = annual; 4 = quarterly;
	12 = monthly; 52 = weekly; 5, 6, or 7 = daily; 24 = hourly).
	In the case of panel data the periodicity means the number of
	lines per data block: this corresponds to the number of
	cross-sectional units in the case of stacked cross-sections,
	or the number of time periods in the case of stacked time series.
	In the case of simple cross-sectional data the periodicity should
	be set to 1.\newline
	The third form of the method (which requires the \gretlcmd{--panel-vars}
	flag) may be used to impose a panel interpretation when the data
	set contains variables that uniquely identify the cross-sectional
	units and the time periods. The data set will be sorted as stacked
	time series, by ascending values of the units variable, \textit{unitvar}.}

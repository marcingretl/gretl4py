\subsection[gretl.Dataset class]{\classpydataset}\label{sec:classpydataset}
This class is designed for loading, storing, and managing
\lgretl['s] \texttt{dataset} type. Thus, it offers attributes
that provide some \texttt{dataset} parameters, and methods
for manipulating data, such as sub-sampling, exporting, filtering, etc.


\subsubsection{\classpydatasetb operators}
The \classpydataset class provides the following operators:

\gretlfpycmd{[]}
\estdesc{Slicing operator for a \classpydataset instance, used to reduce
	its dimension(s).}
\begin{lstlisting}[style=pythonstylefoot]
d[rule: slice | list | str] -> gretl.Dataset
\end{lstlisting}
\where{\pythcodef{rule} is either a sample range, a list of variables,
	or a logical expression that can be applied to the dataset.}\\\\
\generaldesc{Using this operator, a dataset may be sub-sampled along one
	or both dimensions, that is, observations and/or variables. It is
	important to remember that slicing always returns a copy of the
	original dataset.\newline
	Note that nested slicing is supported, which makes the following
	calls equivalent:
	\pythcode{dataset[rule1][rule2]} $\equiv$
	\pythcode{d1 = dataset[rule1]; d2 = d1[rule2]}.
	However, there are two points to keep in mind:
	\begin{itemize}
		\item Since slicing produces a deep copy of the original dataset,
			the one-line expression \pythcode{dataset[rule1][rule2]}
			is more memory-efficient. This is because the intermediate
			temporary object is deallocated automatically, whereas
			\pythcode{d1 = dataset[rule1]; d2 = d1[rule2]} keeps
			\pythcode{d1} alive until it is explicitly deleted.
		\item When slicing along both dimensions, the order of slicing
			matters for efficiency. Because each slicing operation
			creates a copy, the first rule should be the one that
			produces the largest reduction in data, so that less data
			needs to be copied in subsequent operations.
	\end{itemize}}

\gretlfpycmd{bkfilt()}
\estdesc{Returns the result of applying the Baxter--King

\end{lstlisting}
\maptof{bkfilt(y, f1, f2, k)}{series}
\generaldesc{The optional parameters \textit{f1} and
	\textit{f2} specify the lower and upper bounds,
	respectively, of the frequency range to extract, while
	\textit{k} specifies the approximation order. If omitted,
	their default values are 2, 8 and 3 for annual data; 6, 32

\gretlfpycmd{boxplot()}
\estdesc{Generates boxplots for dataset series using \lgretl and Matplotlib.}
\begin{lstlisting}[style=pythonstylefoot]
.boxplot(varlist: str | list[, OPTS]) -> matplotlib.figure.Figure
\end{lstlisting}
\mapto[None]{boxplot ~varlist}
\mapto[None]{boxplot ~varlist --notches}
\mapto[None]{boxplot ~varlist --panel}
\mapto{boxplot ~varlist --factorized}
\begin{itemize}\setlength{\itemsep}{0.3em}\setlength{\parskip}{0pt}\setlength{\parsep}{0pt}
	\item[]\maptoopt{notches=True}{--notches}{show a 90 percent confidence interval for the median}
	\item[]\maptoopt{factorized=True}{--factorized}{produce boxplots conditional on a discrete factor}
	\item[]\maptoopt{panel=True}{--panel}{produce separate boxplots for each panel unit}
\end{itemize}
Examples:\newline
\pythcodef{fig = data.boxplot("y")}\\
\pythcodef{fig = data.boxplot("y1 y2 y3")}\\
\pythcodef{fig = data.boxplot(["y1", "y2"])}\\
\pythcodef{fig = data.boxplot("wage gender", factorized=True)}\\
\pythcodef{fig = data.boxplot("y", panel=True, notches=True)}\\
\generaldesc{Generates boxplots representing the distribution of one
	or more dataset series. Each box extends from the first quartile
	to the third quartile, enclosing the middle 50 percent of the
	observations. The horizontal line inside the box indicates the
	median, while a plus sign denotes the mean.
	
	The whiskers extend up to 1.5 times the interquartile range
	beyond the first and third quartiles. Observations outside
	this range are treated as outliers and displayed as individual
	points.
	
	Setting \textit{notches=True} adds a bootstrap-based 90 percent
	confidence interval for the median, represented by dashed
	horizontal lines.
	
	The \textit{factorized} option produces separate boxplots for
	each distinct value of a discrete factor variable. In this
	case, \textit{varlist} must contain exactly two variables:
	the target variable followed by the factor variable.
	
	For panel datasets, setting \textit{panel=True} produces a
	separate boxplot for each panel unit. This option requires
	exactly one variable in \textit{varlist}.
	
	Plot generation is performed by \lgretl, while the resulting
	\gnuplot script is translated into Python code by \gptranslate
	and rendered using Matplotlib. The method returns a Matplotlib
	Figure object.}

\end{lstlisting}
\mapto[none]{dataset compact ~freq }
\begin{itemize}\setlength{\itemsep}{0.3em}\setlength{\parskip}{0pt}\setlength{\parsep}{0pt}
	\item[] \maptoopt{repday=n}{--repday=n}{representative day for daily-to-weekly compaction}

\generaldesc{Compacts all series in a time-series dataset to
	the specified frequency. By default, higher-frequency values
	are averaged. The optional \textit{method} parameter specifies
	an alternative aggregation method and takes one of the following

	Alternatively, \pythcode{weekstart} specifies the start of the week:

	for 5- or 6-day data the week is taken to start on Monday.}

\gretlfpycmd{hurst()}
\estdesc{Estimates the Hurst exponent for a time series.}
\begin{lstlisting}[style=pythonstylefoot]
.hurst(series: str) -> gretl.Matrix
\end{lstlisting}
\mapto[clean]{hurst ~series}
\generaldesc{The Hurst exponent is discussed by Mandelbrot (1983).
	In theoretical terms it is the exponent, \textit{H}, in the relationship
	\[
		RS(x) = an^H,
	\]
	where \textit{RS} is the ``rescaled range'' of the variable
	\textit{x} in samples of size \textit{n} and \textit{a} is
	a constant. In gretl, the exponent is estimated using binary
	sub-sampling: we start with the entire data range, then the
	two halves of the range, then the four quarters, and so on.
	For sample sizes smaller than the data range, the \textit{RS}
	value is the mean across the available samples. The exponent
	is then estimated as the slope coefficient in a regression of
	the log of \textit{RS} on the log of sample size.
	The returned \classpymatrix contains the estimated Hurst
	exponent and its standard error.}

\gretlfpycmd{xtab()}
\estdesc{Computes cross-tabulations of discrete variables.}
\begin{lstlisting}[style=pythonstylefoot]
.xtab(varlist: str = ''[, OPTS]) -> gretl.Matrix
.xtab(varlist: list = [][, OPTS]) -> gretl.Matrix
\end{lstlisting}
\mapto{xtab ylist [ ; xlist ]}
\begin{itemize}\setlength{\itemsep}{0.3em}\setlength{\parskip}{0pt}\setlength{\parsep}{0pt}
	\item[]\maptoopt{row=True}{--row}{display row percentages}
	\item[]\maptoopt{column=True}{--column}{display column percentages}
	\item[]\maptoopt{zeros=True}{--zeros}{display zero entries}
	\item[]\maptoopt{no\_totals=True}{--no-totals}{suppress marginal counts}
	\item[]\maptoopt{matrix=matname}{--matrix=matname}{use frequencies from a named matrix}
	\item[]\maptoopt{quiet=True}{--quiet}{suppress printed output}
	\item[]\maptoopt{tex=filename}{--tex[=filename]}{produce \LaTeX{} output}
	\item[]\maptoopt{equal=True}{--equal}{highlight equal values in \LaTeX{} output}
\end{itemize}
\generaldesc{Given a single list, computes a cross-tabulation for
	each combination of the variables in \textit{varlist}. If two
	lists are supplied, separated by a semicolon, each variable in
	the first list is cross-tabulated by row against each variable
	in the second list by column. All variables must be marked as
	discrete.\newline
	By default, cells contain frequency counts; these may be replaced
	by row or column percentages using the corresponding options.
	Pearson's chi-square test for independence is reported when the
	expected frequencies satisfy the required minimum condition; for
	a $2\times2$ table, Fisher's exact test is also reported.\newline
	The method returns the cross-tabulation results as a
	\classpymatrix.}
